local function b64decode(data)
    local chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    local lookup = {}
    for i = 1, 64 do lookup[chars:sub(i, i)] = i - 1 end
    data = (data:gsub("[^%w%+/=]", ""))
    local out = {}
    for i = 1, #data, 4 do
        local a, b = lookup[data:sub(i, i)], lookup[data:sub(i + 1, i + 1)]
        local c, d = lookup[data:sub(i + 2, i + 2)], lookup[data:sub(i + 3, i + 3)]
        local v = a * 262144 + b * 4096 + (c or 0) * 64 + (d or 0)
        local s = string.char(math.floor(v / 65536))
        if c then s = s .. string.char(math.floor(v / 256) % 256) end
        if d then s = s .. string.char(v % 256) end
        out[#out + 1] = s
    end
    return table.concat(out)
end

-- only load in games with a code box
do
    local function hasBox(root)
        if not root then return false end
        local codes = root:FindFirstChild("Codes")
        if codes and codes:FindFirstChildWhichIsA("TextBox", true) then return true end
        for _, g in ipairs(root:GetChildren()) do
            if g:IsA("ScreenGui") and not g.Name:find("XENON", 1, true) then
                for _, d in ipairs(g:GetDescendants()) do
                    if d:IsA("TextBox") then
                        local n = (d.Name .. (d.Parent and d.Parent.Name or "") .. g.Name):lower()
                        if n:find("code", 1, true) or n:find("redeem", 1, true) then return true end
                    end
                end
            end
        end
        return false
    end
    local found, deadline = false, os.clock() + 8
    repeat
        found = hasBox(PlayerGui) or hasBox(StarterGui)
        if not found then task.wait(0.25) end
    until found or os.clock() > deadline
    if not found then
        warn("[Xenon Auto Redeemer] No code box in this game - not loading.")
        pcall(function()
            StarterGui:SetCore("SendNotification", { Title = "Xenon Auto Redeemer", Text = "No code box in this game.", Duration = 5 })
        end)
        return
    end
end

    local old = PlayerGui:FindFirstChild("XENON_AUTO")
    if old then old:Destroy() end
end

local function new(class, props, kids)
    local o = Instance.new(class)
    local parent
    for k, v in pairs(props or {}) do
        if k == "Parent" then parent = v else o[k] = v end
    end
    for _, k in ipairs(kids or {}) do k.Parent = o end
    if parent then o.Parent = parent end
    return o
end

    local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function press(btn)
    local sc = btn:FindFirstChildOfClass("UIScale") or new("UIScale", { Parent = btn })
    btn.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            tween(sc, 0.08, { Scale = 0.93 }, Enum.EasingStyle.Quad)
        end
    end)
    btn.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            tween(sc, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
        end
    end)
end

local function makePage(key, scrolling)
    local pg
    if scrolling then
        pg = new("ScrollingFrame", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0,
            CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollBarThickness = 2,

    else
        pg = new("Frame", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Visible = false, ZIndex = 3, Parent = pagesHolder })
    end
    pages[key] = pg
    return pg
end
local function styleTabs()
    for _, t in ipairs(tabNodes) do
        local on = t.key == current
        tween(t.planet, 0.25, { Size = UDim2.fromOffset(on and 30 or 24, on and 30 or 24) }, Enum.EasingStyle.Back)
        t.ringStroke.Transparency = on and 0 or 1
        t.ringStroke.Color = C.accent
        tween(t.label, 0.2, { TextColor3 = on and C.white or C.dim })
        t.glow.BackgroundTransparency = on and 0.75 or 1
    end
end
local function showPage(key)
    if current == key then return end
    local old = current and pages[current]
    current = key
    styleTabs()
    if old then old.Visible = false end
    local pg = pages[key]
    pg.Visible = true
    pg.Position = UDim2.fromOffset(0, 12)
    tween(pg, 0.28, { Position = UDim2.fromOffset(0, 0) })
end

local function chip(parent, order, text, isToggle, initial, onChange)
    local b = new("TextButton", { Text = "", AutoButtonColor = false, BackgroundColor3 = C.panel,
        BorderSizePixel = 0, LayoutOrder = order, ZIndex = 3, Parent = parent })
    corner(b, 12)
    local e = stroke(b, C.line, 1.2, 0)
    local fill = regGrad(new("UIGradient", { Color = ColorSequence.new(C.violet, C.blue), Enabled = false, Parent = b }))
    local ind = new("Frame", { Size = UDim2.fromOffset(8, 8), AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 11, 0.5, 0),
        BackgroundColor3 = C.dim, BorderSizePixel = 0, ZIndex = 4, Parent = b })
    round(ind)
    if not isToggle then ind.BackgroundColor3 = C.accent end
    local t = label(b, { Text = text, Font = GOTHAM_B, TextSize = 11, TextColor3 = C.muted, Position = UDim2.fromOffset(26, 0),
        Size = UDim2.new(1, -30, 1, 0), ZIndex = 4 })
    press(b)
    local state = initial and true or false
    local function paint(v)
        fill.Enabled = v
        b.BackgroundColor3 = v and C.white or C.panel
        tween(t, 0.18, { TextColor3 = v and C.white or C.muted })
        tween(ind, 0.25, { BackgroundColor3 = v and C.white or C.dim, Size = v and UDim2.fromOffset(10, 10) or UDim2.fromOffset(8, 8) },
            Enum.EasingStyle.Back)
        e.Color = v and C.accent or C.line
        e.Transparency = v and 0.4 or 0
    end
    if isToggle then paint(state) end
    local api = {}
    function api.set(v, silent)
        state = v
        paint(v)
        if onChange and not silent then onChange(v) end
    end
    function api.get() return state end
    b.MouseButton1Click:Connect(function()
        if isToggle then
            api.set(not state)
        else
            local g = new("Frame", { Size = UDim2.fromScale(1, 1), BackgroundColor3 = C.violet, BackgroundTransparency = 0.5,
                BorderSizePixel = 0, ZIndex = 3, Parent = b })
            corner(g, 12)
            tween(g, 0.4, { BackgroundTransparency = 1 })
            Debris:AddItem(g, 0.45)
            if onChange then onChange() end
        end
    end)
    return api
end

local function refreshIdle()
    if S.listening then
        setStatus(("listening  ·  %d/%d"):format(#S.buffer, S.words), C.yellow)
    else
        setStatus(UserInputService.KeyboardEnabled and "idle  ·  press listen key or tap Listen" or "idle  ·  tap Listen", C.red)
    end
    wordHint.Text = S.listening and ("%d / %d words"):format(#S.buffer, S.words) or ""
end

local function selectWords(n, fromCode)
    n = math.clamp(n, 1, 6)
    S.words = n
    tween(pill, 0.38, { Position = UDim2.fromOffset(3 + (n - 1) * 34, 3) }, Enum.EasingStyle.Back)
    pillScale.Scale = 0.8
    tween(pillScale, 0.35, { Scale = 1 }, Enum.EasingStyle.Back)
    for i, b in ipairs(wordBtns) do tween(b, 0.2, { TextColor3 = (i == n) and C.white or C.muted }) end
    if fromCode then
        local lbl = wordsRow:FindFirstChildOfClass("TextLabel")
        if lbl then
            lbl.TextColor3 = C.green
            tween(lbl, 1.2, { TextColor3 = C.accent })
        end
    end
    refreshIdle()
end

function R.findBox()

local TW = { target = nil, running = false, waiters = {} }
function R.typeTo(target, onDone)
    local box = R.findBox()
    TW.target = target
    if onDone then TW.waiters[#TW.waiters + 1] = { target, onDone } end
    if TW.running then return end
    TW.running = true
    S.typing = true
    task.spawn(function()
        local shown = codeText.Text
        while TW.target do
            local tgt = TW.target
            if shown == tgt then
                local keep = {}
                for _, w in ipairs(TW.waiters) do
                    if w[1] == tgt then task.spawn(w[2], box) else keep[#keep + 1] = w end
                end
                TW.waiters = keep
                if TW.target == tgt then break end
            elseif tgt:sub(1, #shown) == shown then
                shown = tgt:sub(1, #shown + 1)
                keyClick()
                task.wait(S.typeDelay)
            else
                shown = shown:sub(1, #shown - 1)
                task.wait(0.012)
            end
            if box and box.Parent then pcall(function() box.Text = shown end) end
            showCode(shown)
        end
        TW.running = false
        S.typing = false
        caret.Visible = false
    end)
end

function R.submit(code, box)
    box = box or R.findBox()
    if not box then setStatus("no code box found", C.red) return false end
    pcall(function() box.Text = code end)
    local btn = R.findButton(box)
    local gc = getconnections or (debug and debug.getconnections)
    local fired = false
    if btn and type(gc) == "function" then
        for _, sig in ipairs({ btn.MouseButton1Click, btn.Activated }) do
            local ok, cs = pcall(gc, sig)
            if ok and type(cs) == "table" and #cs > 0 then
                for _, c in ipairs(cs) do
                    local fn
                    pcall(function() fn = c.Function end)
                    if fn then fired = pcall(task.spawn, fn) or fired else fired = pcall(function() c:Fire() end) or fired end
                end
                if fired then break end
            end
        end
    end
    if not fired and type(firesignal) == "function" then
        if btn then fired = pcall(firesignal, btn.MouseButton1Click) or pcall(firesignal, btn.Activated) end
        if not fired then fired = pcall(firesignal, box.FocusLost, true) end
    end
    if not fired then
        fired = pcall(function()
            box:CaptureFocus()
            task.wait()
            box:ReleaseFocus(true)
        end)
    end
    return fired
end

function R.watch(code, box)
    local done, conns = false, {}
    local function finish(ok, msg)
        if done then return end
        done = true
        for _, c in ipairs(conns) do c:Disconnect() end
        if ok == nil then return end
        if ok then
            if S.listening then setListen(false) end
            setStatus("redeemed  " .. code .. "  ✓", C.green)
            codeEdge.Color = C.green
            tween(codeEdge, 1.2, { Color = C.line })
        else
            setStatus(tostring(msg):gsub("<[^>]->", ""):sub(1, 40) .. "  ·  " .. code, C.red)
        end
    end
    local function check(t)
        if type(t) ~= "string" or t == "" or t == code then return end
        local low = t:gsub("<[^>]->", ""):lower()
        for _, pr in ipairs(RESULT) do
            if low:find(pr[1], 1, true) then finish(pr[2], t) return end
        end
    end
    if box then
        conns[#conns + 1] = box:GetPropertyChangedSignal("Text"):Connect(function() check(box.Text) end)
        conns[#conns + 1] = box:GetPropertyChangedSignal("PlaceholderText"):Connect(function() check(box.PlaceholderText) end)
    end
    conns[#conns + 1] = PlayerGui.DescendantAdded:Connect(function(d)
        if (d:IsA("TextLabel") or d:IsA("TextButton")) and not d:IsDescendantOf(gui) then
            task.defer(function() if d.Parent then check(d.Text) end end)
        end
    end)
    task.delay(4, function() finish(nil) end)
end

local TRIGGERS = { "use%s+codes?", "codes?%s+is", "codes?%s+are", "the%s+code", "new%s+code", "codes?%s*:", "redeem", "enter%s+code", "type%s+code" }
local function triggerEnd(lo)
    local last
    for _, pat in ipairs(TRIGGERS) do
        local init = 1
        while true do
            local a, b = lo:find(pat, init)
            if not a then break end
            if not last or b > last then last = b end
            init = b + 1
        end
    end
    return last
end
local function finishCode(words)
    S.lastWords = words
    local code = table.concat(words)
    setListen(false)
    R.redeem(code)
end
local function handle(raw)
    local text = raw:gsub("<[^>]->", ""):gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return end
    local lo = text:lower()
    local multi = text:find(" ", 1, true) ~= nil
    if S.ignoreSystem and multi and not lo:find("code", 1, true) then
        for _, w in ipairs(SYS) do if lo:find(w, 1, true) then return end end
    end
    if lo:find("code", 1, true) then
        local hn = tonumber(lo:match("(%d+)%s*%-?%s*words?%s+codes?")
            or ({ one = "1", two = "2", three = "3", four = "4", five = "5", six = "6" })[lo:match("(%a+)%s*%-?%s*words?%s+codes?") or ""])
        if hn then
            selectWords(hn, true)
            setStatus(("heads up  ·  %d-word code"):format(hn), C.green)
            return
        end
        local cut = triggerEnd(lo)
        if cut then
            local words = {}
            for w in text:sub(cut + 1):gmatch("[%w_]+") do
                if not FILLER[w:lower()] then words[#words + 1] = w end
            end
            if #words == 0 then
                if S.autoListen and not S.listening then setListen(true) end
                if S.listening then
                    S.buffer = {}
                    setStatus("trigger heard  ·  waiting for the code…", C.accent)
                end
                return
            end
            if S.listening or S.autoListen then finishCode(words) end
            return
        end
    end
    if not S.listening then return end
    for w in text:gmatch("[%w_]+") do
        S.buffer[#S.buffer + 1] = w
        if #S.buffer >= S.words then
            local words = {}
            for i = 1, S.words do words[i] = S.buffer[i] end
            finishCode(words)
            return
        end
        if S.twoWord and #S.buffer == 2 and S.words > 2 then
            local two = S.buffer[1] .. S.buffer[2]
            setStatus(("2-word check  ·  trying %s"):format(two), C.accent)
            R.redeem(two)
        end
    end
    local partial = table.concat(S.buffer)
    R.typeTo(partial)
    refreshIdle()
end

-- drag by the header
do
    local dragging, start, origin
    head.InputBegan:Connect(function(i)
        if i.UserInputType ~= Enum.UserInputType.MouseButton1 and i.UserInputType ~= Enum.UserInputType.Touch then return end
        dragging, start, origin = true, i.Position, holder.Position
        i.Changed:Connect(function() if i.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end)
    UserInputService.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType ~= Enum.UserInputType.MouseMovement and i.UserInputType ~= Enum.UserInputType.Touch then return end
        local d = i.Position - start
        holder.Position = UDim2.new(origin.X.Scale, origin.X.Offset + d.X, origin.Y.Scale, origin.Y.Offset + d.Y)
    end)
end

-- keys: RightShift show/hide, listen key toggles listen
UserInputService.InputBegan:Connect(function(i, processed)
    if processed then return end
    if i.KeyCode == Enum.KeyCode.RightShift then
        holder.Visible = not holder.Visible
    elseif S.listenKey and i.KeyCode == S.listenKey then
        setListen(not S.listening)
    end
end)

-- one animation loop
local sweepT = 0
RunService.RenderStepped:Connect(function(dt)
    if not holder.Visible then return end
    dt = math.min(dt, 1 / 20)
    local t = os.clock()
    local H = card.AbsoluteSize.Y / math.max(fit.Scale, 0.01)

    for _, s in ipairs(stars) do
        s.x -= s.v * dt * 0.6
        s.y += s.v * dt
        if s.y > H + 4 or s.x < -4 then s.y, s.x = -4, SRNG:NextNumber(0, WIN + 60) end
        s.f.Position = UDim2.fromOffset(s.x, s.y)
        s.f.BackgroundTransparency = math.clamp(s.base + math.sin(t * s.tw + s.ph) * 0.25, 0, 0.95)
    end
    for _, sh in ipairs(shooters) do
        if sh.life > 0 then
            sh.life -= dt
            sh.x += 300 * dt
            sh.y += 160 * dt
            sh.f.Position = UDim2.fromOffset(sh.x, sh.y)
            sh.f.BackgroundTransparency = 1 - math.clamp(sh.life / 0.7, 0, 1)
            if sh.life <= 0 then sh.f.Visible = false end
        else
            sh.next -= dt
            if sh.next <= 0 then
                sh.next = SRNG:NextNumber(4, 9)
                sh.life = 0.7
                sh.x, sh.y = SRNG:NextNumber(-20, WIN * 0.5), SRNG:NextNumber(-10, 80)
                sh.f.Visible = true
            end
        end
    end
    if petalsOn then
        for _, p in ipairs(petals) do
            p.y += p.v * dt
            p.x += math.sin(t * 0.8 + p.ph) * p.sw * dt
            if p.y > H + 12 then p.y, p.x = -8, SRNG:NextNumber(0, WIN) end
            p.rot += p.spin * dt
            p.f.Position = UDim2.fromOffset(p.x, p.y)
            p.f.Rotation = p.rot
        end
    end
    edgeGrad.Rotation = (t * 30) % 360

    local speed = S.listening and 7 or 4
    for i, d in ipairs(dots) do
        if d.Visible then
            local y = math.sin(t * speed - i * 0.7) * (S.listening and 7 or 5)
            d.Position = UDim2.new(0.5, (i - 2) * 18, 0.5, -y)
            d.BackgroundColor3 = S.listening and C.accent or C.muted
            d.BackgroundTransparency = 0.25 - math.max(0, math.sin(t * speed - i * 0.7)) * 0.25
        end
    end

    if S.listening then
        sweepT = (sweepT + dt * 0.7) % 1.4
        sweep.Position = UDim2.new(sweepT - 0.4, 0, 0, 0)
        codeEdgeGrad.Rotation = (t * 90) % 360
        local b = (math.sin(t * 3) + 1) / 2
        statusHalo.BackgroundColor3 = statusDot.BackgroundColor3
        statusHalo.Size = UDim2.fromOffset(12 + b * 6, 12 + b * 6)
        statusHalo.BackgroundTransparency = 0.55 + 0.35 * (1 - b)
    elseif statusHalo.BackgroundTransparency < 1 then
        statusHalo.BackgroundTransparency = 1
    end

    -- themes page: twinkle the preview dots
    if pThemes.Visible then
        for _, tc in ipairs(themeCards) do
            tc.tw1.BackgroundTransparency = math.clamp(0.2 + math.sin(t * 2 + tc.th.c1.R * 6) * 0.4, 0, 0.9)
            tc.tw2.BackgroundTransparency = math.clamp(0.2 + math.sin(t * 2.6 + tc.th.c2.G * 6) * 0.4, 0, 0.9)
        end
    end

    if caret.Visible then
        local w = codeText.TextBounds.X / math.max(fit.Scale, 0.01)
        caret.Position = UDim2.new(0.5, math.min(w / 2 + 3, codeBox.AbsoluteSize.X / fit.Scale / 2 - 10), 0.5, 0)
        caret.BackgroundTransparency = (math.floor(t * 3) % 2 == 0) and 0 or 0.8
    end
end)
